## 1. 文件工具 `src/utils/file.util.ts`

对应 nest-admin 的 `src/utils/file.util.ts`。它从 fastify 的 `MultipartFile` 换成了 multer 的 Buffer：

```ts
import fs from 'node:fs'
import path from 'node:path'

export enum FileType {
  IMAGE = '图片',
  TXT = '文档',
  MUSIC = '音乐',
  VIDEO = '视频',
  OTHER = '其他'
}

/** 按扩展名归类文件类型 */
export function getFileType(extName: string): FileType {
  const documents = 'txt doc pdf ppt pps xlsx xls docx'
  const music = 'mp3 wav wma mpa ram ra aac aif m4a'
  const video = 'avi mpg mpe mpeg asf wmv mov qt rm mp4 flv m4v webm ogv ogg'
  const image = 'bmp dib pcp dif wmf gif jpg tif eps psd cdr iff tga pcd mpt png jpeg'
  if (image.includes(extName)) return FileType.IMAGE
  if (documents.includes(extName)) return FileType.TXT
  if (music.includes(extName)) return FileType.MUSIC
  if (video.includes(extName)) return FileType.VIDEO
  return FileType.OTHER
}

/** 取不含点的小写扩展名 */
export function getExtname(fileName: string): string {
  return path.extname(fileName).replace('.', '').toLowerCase()
}

/** 字节数转可读大小，如 "1.5 MB" */
export function getSize(bytes: number, decimals = 2): string {
  if (bytes === 0) return '0 B'
  const k = 1024
  const sizes = ['B', 'KB', 'MB', 'GB', 'TB']
  const i = Math.floor(Math.log(bytes) / Math.log(k))
  return `${Number.parseFloat((bytes / k ** i).toFixed(Math.max(decimals, 0)))} ${sizes[i]}`
}

/** 重命名：原名 + 时间戳 + 扩展名，防覆盖 */
export function fileRename(fileName: string): string {
  const name = fileName.split('.')[0]
  const extName = path.extname(fileName)
  const time = new Date().toISOString().replace(/\D/g, '').slice(0, 17)
  return `${name}-${time}${extName}`
}

/** 数据库中存的相对访问路径 */
export function getFilePath(name: string, currentDate: string, type: string): string {
  return `/upload/${currentDate}/${type}/${name}`
}

/** 当前日期目录名 YYYY-MM-DD */
export function getCurrentDate(): string {
  const d = new Date()
  const pad = (n: number) => String(n).padStart(2, '0')
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
}

/** 把 Buffer 落盘到 public/upload/{日期}/{类型}/ 目录 */
export async function saveLocalFile(buffer: Buffer, name: string, currentDate: string, type: string): Promise<void> {
  const dir = path.join(process.cwd(), 'public/upload', currentDate, type)
  await fs.promises.mkdir(dir, { recursive: true })
  await fs.promises.writeFile(path.join(dir, name), buffer)
}

/** 按相对访问路径删除本地文件（失败静默，不阻断主流程） */
export function deleteFile(relativePath: string): void {
  fs.unlink(path.join(process.cwd(), 'public', relativePath), () => {})
}
```

**学习点**：
- 目录按 `日期/类型` 二级分桶，避免单目录文件爆炸；`mkdir recursive: true` 一步建多级
- 落盘用 `fs.promises`（async），删除用回调式且吞错——删文件失败不应影响删除记录的主流程
- nest-admin 原版用 `createWriteStream.write(buffer)`，其实 buffer 已在内存，直接 `writeFile` 更简洁

## 3. 上传服务 `src/modules/tools/upload/upload.service.ts`

```ts
import { Injectable, NotFoundException } from '@nestjs/common'

import { StorageEntity } from '../storage/storage.entity'
import { StorageService } from '../storage/storage.service'
import { fileRename, getExtname, getCurrentDate, getFilePath, getFileType, getSize, saveLocalFile } from '../../../utils/file.util'

/**
 * 上传服务：文件重命名落盘 public/upload/{日期}/{类型}/，并写入 tool_storage 记录
 * 参照 nest-admin 的 tools/upload 结构，multer 文件由控制器传入
 */
@Injectable()
export class UploadService {
  constructor(private readonly storageService: StorageService) {}

  /** 保存一个上传文件，返回相对访问路径（静态目录挂载在 /public） */
  async saveFile(file: Express.Multer.File | undefined, userId?: number): Promise<{ path: string }> {
    if (!file) {
      throw new NotFoundException('未找到上传文件')
    }
    const fileName = file.originalname
    const size = getSize(file.size)
    const extName = getExtname(fileName)
    const type = getFileType(extName)
    const name = fileRename(fileName)
    const currentDate = getCurrentDate()
    const path = getFilePath(name, currentDate, type)

    await saveLocalFile(file.buffer, name, currentDate, type)

    const record = new StorageEntity()
    record.name = name
    record.fileName = fileName
    record.extName = extName
    record.path = path
    record.type = type
    record.size = size
    record.userId = userId
    await this.storageService.save(record)

    return { path }
  }
}
```

**学习点**：
- multer（内存存储默认策略）把文件读成 `file.buffer`，与 fastify 的 `await file.toBuffer()` 对应
- 职责链：控制器只取文件 → service 组装元数据 → file.util 做纯函数操作，service 不直接碰 fs

## 4. 上传控制器 `src/modules/tools/upload/upload.controller.ts`

```ts
import { Controller, Post, Request, UploadedFile, UseInterceptors } from '@nestjs/common'
import { FileInterceptor } from '@nestjs/platform-express'
import { ApiBearerAuth, ApiBody, ApiConsumes, ApiOkResponse, ApiOperation, ApiTags } from '@nestjs/swagger'

import { Perms } from '../../../common/decorators/perms.decorator'
import { FileUploadDto } from './upload.dto'
import { UploadService } from './upload.service'

/** 与参考项目一致：仅图片，上限 10MB */
export const UPLOAD_FILE_SIZE = 10 * 1024 * 1024
export const UPLOAD_MIMETYPES = ['image/png', 'image/gif', 'image/jpeg', 'image/webp', 'image/svg+xml']

@ApiTags('工具 - 上传模块')
@ApiBearerAuth('jwt')
@Controller('upload')
export class UploadController {
  constructor(private readonly uploadService: UploadService) {}

  @ApiOperation({ summary: '上传文件', description: 'multipart/form-data，字段名 file；仅图片，上限 10MB' })
  @ApiConsumes('multipart/form-data')
  @ApiBody({ type: FileUploadDto })
  @ApiOkResponse({ description: '返回文件相对路径 path，静态访问地址为 {host}{path}' })
  @Perms('tools:upload')
  @Post()
  @UseInterceptors(FileInterceptor('file', { limits: { fileSize: UPLOAD_FILE_SIZE } }))
  upload(@UploadedFile() file: Express.Multer.File, @Request() req: { user?: { id: number } }) {
    return this.uploadService.saveFile(file, req.user?.id)
  }
}
```

**学习点**：
- nest-admin 是 fastify 的 `req.file()`，Express 这边用 `FileInterceptor('file')` + `@UploadedFile()`，multer 随 `@nestjs/platform-express` 自带，无需额外安装运行时依赖
- 超限文件 multer 直接报错（.multer 对象），由全局异常过滤器兜底
- `req.user` 来自全局 JwtAuthGuard 写入，取 `id` 记录上传人

## 5. 存储服务 `src/modules/tools/storage/storage.service.ts`

```ts
import { Injectable, Logger, NotFoundException } from '@nestjs/common'
import { InjectRepository } from '@nestjs/typeorm'
import { In, Repository } from 'typeorm'

import { deleteFile } from '../../../utils/file.util'
import { paginate, type Pagination } from '../../../helper/paginate'
import { StorageEntity } from './storage.entity'

/**
 * 文件存储服务：tool_storage 记录的查询与删除
 * 删除时同步清理本地文件，文件丢失不阻断记录删除
 */
@Injectable()
export class StorageService {
  private readonly logger = new Logger(StorageService.name)

  constructor(
    @InjectRepository(StorageEntity)
    private readonly storageRepository: Repository<StorageEntity>
  ) {}

  /** 保存一条上传记录 */
  async save(record: StorageEntity): Promise<StorageEntity> {
    return this.storageRepository.save(record)
  }

  /** 分页查询存储记录，支持文件名/类型过滤 */
  async list(query: { page?: number; pageSize?: number; name?: string; type?: string }): Promise<Pagination<StorageEntity>> {
    const qb = this.storageRepository.createQueryBuilder('storage').orderBy('storage.id', 'DESC')
    if (query.name) {
      qb.andWhere('storage.name LIKE :name', { name: `%${query.name}%` })
    }
    if (query.type) {
      qb.andWhere('storage.type = :type', { type: query.type })
    }
    return paginate(qb, { page: query.page, pageSize: query.pageSize })
  }

  /** 批量删除记录并清理本地文件，返回删除条数 */
  async remove(ids: number[]): Promise<{ deleted: number }> {
    const records = await this.storageRepository.find({ where: { id: In(ids) } })
    for (const record of records) {
      deleteFile(record.path)
    }
    const result = await this.storageRepository.delete({ id: In(ids) })
    if (records.length !== ids.length) {
      this.logger.warn(`存储记录删除请求 ${ids.length} 条，实际存在 ${records.length} 条`)
    }
    return { deleted: result.affected ?? 0 }
  }

  /** 存储记录详情，不存在抛 404 */
  async findOne(id: number): Promise<StorageEntity> {
    const record = await this.storageRepository.findOne({ where: { id } })
    if (!record) {
      throw new NotFoundException('文件记录不存在')
    }
    return record
  }
}
```

**学习点**：删除前先 `find` 拿到 path 才能删本地文件；数量对不上打 warn——记录可能已被并发删掉。

    const rows: Array<{ id: number }> = await queryRunner.query(`SELECT id FROM menu WHERE name = ? AND type = ? ORDER BY id ASC LIMIT 1`, [name, type])
    return rows.length > 0 ? rows[0].id : null
  }
}

## 遇到的坑

1. **tsconfig types 限定**：`"types": ["node","jest"]` 把 @types/multer 的全局 `Express.Multer` 命名空间挡掉了，报 `Namespace 'Express' has no exported member 'Multer'`——把 `"multer"` 加进 types 数组解决。
2. **useStaticAssets 不在 INestApplication 上**：需要 `NestFactory.create<NestExpressApplication>(...)`。
3. **静态路径含中文目录（/图片/）**：HTTP 请求行不允许裸中文，浏览器自动百分号编码而 curl 不会，手测 400 别误判为服务问题。
4. **Windows 端口残留**：taskkill 杀 node 子进程后父进程可能仍占 3459（EADDRINUSE），用 `netstat -ano | grep LISTENING` 找所有 PID 逐一杀。

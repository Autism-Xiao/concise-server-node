## 背景

用户参照自己整理的「统一响应格式 + BizErrorCode + BizException + GlobalExceptionFilter」模板，要求：

1. **补充业务错误码**：验证码细分（412/413/414/415/416/417），参数/鉴权/业务类用 1xxx 段（1001~1006），登录失败扩展为 1007、账号停用 1008；替换掉旧的 `BusinessException`。

/**
 * 业务错误码（响应体 code 字段）
 *
 * 约定：
 * - 0 表示成功，由 TransformInterceptor 统一返回
 * - 4xx 段沿用 HTTP 语义的数字（412/413/414/415/416/417），方便与 JeecgBoot 前端约定对齐
 * - 1xxx 段为本项目自定义业务码，避免与 HTTP 状态码混淆
 * - 前端判断成功只看 success 或 code === 0，不要用 HTTP 状态码
 */
export enum BizErrorCode {
  /** 验证码错误 */
  CAPTCHA_INVALID = 412,
  /** 验证码已过期 */
  CAPTCHA_EXPIRED = 413,
  /** 验证码不存在 */
  CAPTCHA_NOT_FOUND = 414,
  /** 验证码已被使用 */
  CAPTCHA_ALREADY_USED = 415,
  /** 验证码发送过于频繁 */
  CAPTCHA_SEND_TOO_FREQUENT = 416,
  /** 验证码参数错误 */
  CAPTCHA_PARAM_INVALID = 417,

  /** 参数校验失败 */
  PARAM_INVALID = 1001,
  /** 资源不存在 */
  NOT_FOUND = 1002,
  /** 未授权 */
  UNAUTHORIZED = 1003,
  /** 禁止访问 */
  FORBIDDEN = 1004,
  /** 业务处理失败 */
  BUSINESS_ERROR = 1005,
  /** 服务器内部错误 */
  INTERNAL_ERROR = 1006,

  /** 登录失败（用户名或密码错误）—— 本项目扩展 */
  LOGIN_FAILED = 1007,
  /** 账号已停用/冻结 —— 本项目扩展 */
  ACCOUNT_DISABLED = 1008,

  /** 依赖服务不可用（如 Redis 未启用导致验证码不可用），沿用原 503 保持兼容 */
  SERVICE_UNAVAILABLE = 503
}

import { HttpException, HttpStatus } from '@nestjs/common'

import { BizErrorCode } from '../enums/biz-error-code.enum'

/**
 * 业务异常：带业务错误码，由 AllExceptionsFilter 统一组装成标准 Result。
 *
 * 为什么默认 HTTP 200 而不是 400：
 * 本项目的约定是「业务失败也走 HTTP 200，用 code 区分」，前端 axios 不会进 catch 分支，
 * 统一在 then 里按 success / code 处理，避免「同一个错误弹两个提示」。
 * 确实需要非 200 语义的（如登录失败想保留 401、账号停用用 403），显式传第三个参数即可。
 *
 * 用法：
 * ```ts
 * throw new BizException(BizErrorCode.CAPTCHA_INVALID, '验证码错误')
 * throw new BizException(BizErrorCode.LOGIN_FAILED, '用户名或密码错误', HttpStatus.UNAUTHORIZED)
 * ```
 */
export class BizException extends HttpException {
  constructor(
    public readonly bizCode: BizErrorCode,
    message: string,
    httpStatus: HttpStatus = HttpStatus.OK
  ) {
    // 这里把结构先塞进 response，过滤器可直接 exception.getResponse() 复用（也可只用 bizCode）
    super({ success: false, code: bizCode, message, result: null }, httpStatus)
  }

  getBizCode(): BizErrorCode {
    return this.bizCode
  }
}
```

### 3. `src/common/filters/any-exception.filter.ts`（重写）

```ts
import { type ArgumentsHost, Catch, type ExceptionFilter, HttpException, HttpStatus } from '@nestjs/common'
import type { Response } from 'express'

import { winstonLogger } from '../../shared/logger/winston.logger'
import { BizErrorCode } from '../enums/biz-error-code.enum'
import { BizException } from '../exceptions/biz.exception'
import type { ApiResult } from '../interfaces/api-result.interface'

interface ExceptionResponse {
  message: string | string[]
  code?: number
}

/** 常见 HTTP 状态码 → 业务错误码 */
const HTTP_STATUS_TO_BIZ_CODE: Record<number, BizErrorCode> = {
  [HttpStatus.BAD_REQUEST]: BizErrorCode.PARAM_INVALID,
  [HttpStatus.UNAUTHORIZED]: BizErrorCode.UNAUTHORIZED,
  [HttpStatus.FORBIDDEN]: BizErrorCode.FORBIDDEN,
  [HttpStatus.NOT_FOUND]: BizErrorCode.NOT_FOUND
}

/** Nest 内置异常默认的英文提示，出现时替换成中文 */
const DEFAULT_EN_MESSAGES: Record<number, string> = {
  [HttpStatus.UNAUTHORIZED]: 'Unauthorized',
  [HttpStatus.FORBIDDEN]: 'Forbidden',
  [HttpStatus.NOT_FOUND]: 'Not Found'
}

/** 默认中文提示 */
const DEFAULT_CN_MESSAGES: Record<number, string> = {
  [HttpStatus.UNAUTHORIZED]: '未授权，请先登录',
  [HttpStatus.FORBIDDEN]: '禁止访问',
  [HttpStatus.NOT_FOUND]: '资源不存在'
}

/** 组装失败响应（result 恒为 null，success 恒为 false） */
function fail(code: number, message: string): ApiResult<null> {
  return { code, message, result: null, success: false, timestamp: Date.now() }
}

/** 从 HttpException 里取展示文案：class-validator 返回数组，只取第一条 */
function pickMessage(exception: HttpException): string {
  const res = exception.getResponse()
  const raw = typeof res === 'string' ? res : (res as ExceptionResponse).message
  if (Array.isArray(raw)) {
    return raw[0] ?? exception.message
  }
  return raw ?? exception.message
}

/** 英文默认提示换中文；有自定义 message 时原样保留 */
function localize(status: number, message: string): string {
  if (!message || message === DEFAULT_EN_MESSAGES[status]) {
    return DEFAULT_CN_MESSAGES[status] ?? message
  }
  return message
}

@Catch()
export class AllExceptionsFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp()
    const response = ctx.getResponse<Response>()
    const request = ctx.getRequest<{ method: string; url: string }>()

    // 1. 业务异常：code 用枚举值，HTTP 状态由异常自身决定（默认 200）
    if (exception instanceof BizException) {
      return response.status(exception.getStatus()).json(fail(exception.getBizCode(), exception.message))
    }

    // 2. Nest 内置异常：HTTP 状态保留，code 映射成业务码
    if (exception instanceof HttpException) {
      const status = exception.getStatus()
      const code = HTTP_STATUS_TO_BIZ_CODE[status] ?? BizErrorCode.INTERNAL_ERROR
      return response.status(status).json(fail(code, localize(status, pickMessage(exception))))
    }

    // 3. 未知异常：记录完整堆栈，对外只暴露通用提示
    winstonLogger.error(`${request.method} ${request.url} -> ${exception instanceof Error ? exception.stack : String(exception)}`)
    return response.status(HttpStatus.INTERNAL_SERVER_ERROR).json(fail(BizErrorCode.INTERNAL_ERROR, '服务器内部错误'))
  }
}
```

### 4. `src/common/interfaces/api-result.interface.ts`（追加别名）

```ts
/**
 * 统一响应结构（对齐 JeecgBoot 的 Result<T>）
 *
 * 成功：{ code: 0, message: '', result: <业务数据>, success: true, timestamp: 1793768743245 }
 * 失败：{ code: <非 0>, message: '<可直接提示的文案>', result: null, success: false, timestamp: ... }
 *
 * - code 不是 HTTP 状态码：成功恒为 0；业务异常用业务码（如 503），参数/鉴权异常用 HTTP 状态码
 * - success 是冗余的快速判断位，等价于 code === 0，方便前端 `if (res.success)` 直接判断
 * - timestamp 由服务端在包装时生成（毫秒），用于排查前端缓存/时序问题
 */
export interface ApiResult<T = unknown> {
  /** 成功固定 0；失败为业务错误码或 HTTP 状态码 */
  code: number
  /** 成功为空串，失败为可直接展示的错误信息 */
  message: string
  /** 业务数据，失败时为 null */
  result: T | null
  /** 是否成功，等价于 code === 0 */
  success: boolean
  /** 服务端响应时间戳（毫秒） */
  timestamp: number
}

/** 兼容命名：部分脚手架/文档里叫 ApiResponse，与 ApiResult 完全等价 */
export type ApiResponse<T = unknown> = ApiResult<T>
```

### 6. `src/modules/auth/services/captcha.service.ts`（重写）

```ts
import { Injectable } from '@nestjs/common'
import { randomUUID } from 'crypto'
import * as svgCaptcha from 'svg-captcha'

import { BizErrorCode } from '../../../common/enums/biz-error-code.enum'
import { BizException } from '../../../common/exceptions/biz.exception'
import { RedisService } from '../../../shared/redis/redis.service'

/** 验证码在 Redis 中的 key 前缀 */
const CAPTCHA_KEY_PREFIX = 'captcha:'

/**
 * 图形验证码服务（防爆破）
 * 存储只走 Redis（EX 300 自动过期），支持多实例部署；
 * 未启用 Redis（REDIS_ENABLED=false）或 Redis 故障时直接抛业务错误，不做内存降级，
 * 避免多实例间「A 实例生成、B 实例校验」的状态不一致。过期回收交给 Redis EX。
 */
@Injectable()
export class CaptchaService {
  /** 有效期 5 分钟 */
  private readonly ttlSeconds = 300

  constructor(private readonly redisService: RedisService) {}

  /** 生成验证码：返回 SVG 与 captchaId；debug 模式附带明文（仅开发用） */
  async generate(debug = false): Promise<{ captchaId: string; svg: string; code?: string }> {
    const client = this.requireClient()
    const { text, data: svg } = svgCaptcha.create({
      size: 4,
      ignoreChars: 'il1O0o',
      noise: 3,
      color: true,
      background: '#f5f5f5'
    })
    const captchaId = randomUUID()
    await client.set(CAPTCHA_KEY_PREFIX + captchaId, text.toLowerCase(), 'EX', this.ttlSeconds)
    return debug ? { captchaId, svg, code: text } : { captchaId, svg }
  }

  /**
   * 校验并销毁（一次性）：MULTI 事务内 GET+DEL 原子执行，并发重放只有首个请求能命中。
   * 校验不通过直接抛 BizException，错误码细分到「参数缺失 / 已过期 / 已被使用 / 内容错误」。
   */
  async assertValid(captchaId?: string, code?: string): Promise<void> {
    if (!captchaId || !code) {
      throw new BizException(BizErrorCode.CAPTCHA_PARAM_INVALID, '请输入验证码')
    }
    const expected = code.trim().toLowerCase()
    const client = this.requireClient()
    const results = (await client
      .multi()
      .get(CAPTCHA_KEY_PREFIX + captchaId)
      .del(CAPTCHA_KEY_PREFIX + captchaId)
      .exec()) as Array<[null, string | number | null]>
    const [, text] = results[0]
    const [, deleted] = results[1]

    if (typeof text !== 'string') {
      // 值为空：已过期（超过 TTL）或这个 captchaId 从未生成过
      throw new BizException(BizErrorCode.CAPTCHA_EXPIRED, '验证码已过期，请重新获取')
    }
    if (!deleted) {
      // 读到了值却删不掉：说明刚被另一个请求消费掉（并发抢用）
      throw new BizException(BizErrorCode.CAPTCHA_ALREADY_USED, '验证码已被使用，请重新获取')
    }
    if (text !== expected) {
      throw new BizException(BizErrorCode.CAPTCHA_INVALID, '验证码错误，请重新获取')
    }
  }

  /** 未启用 Redis 时统一抛业务错误（HTTP 200 + code 503），由前端统一拦截提示 */
  private requireClient() {
    const client = this.redisService.client
    if (!client) {
      throw new BizException(BizErrorCode.SERVICE_UNAVAILABLE, '验证码服务不可用，请检查 Redis 配置（REDIS_ENABLED）')
    }
    return client
  }
}
```

### 7. `src/modules/auth/services/auth.service.ts`（修改：登录校验换业务码）

```ts
import { Injectable, HttpException, HttpStatus } from '@nestjs/common'
import { JwtService } from '@nestjs/jwt'
import { ConfigService } from '@nestjs/config'
import type { StringValue } from 'ms'

import { BizErrorCode } from '../../../common/enums/biz-error-code.enum'
import { BizException } from '../../../common/exceptions/biz.exception'
import { UserService, USER_STATUS } from '../../user/user.service'
import { LogService } from '../../system/log/log.service'
import { CaptchaService } from './captcha.service'
import { LoginDto } from '../dto/login.dto'

export interface JwtPayload {
  id: string
  username: string
}

export interface LoginResult {
  accessToken: string
  refreshToken: string
  expiresIn: number
  userInfo: Record<string, unknown>
}

@Injectable()
export class AuthService {
  constructor(
    private readonly userService: UserService,
    private readonly jwtService: JwtService,
    private readonly configService: ConfigService,
    private readonly logService: LogService,
    private readonly captchaService: CaptchaService
  ) {}

  /** 登录：验证码校验（可开关）→ 账号密码校验 → 签发双令牌；成败都记入登录日志 */
  async login(dto: LoginDto, meta: { ip: string; userAgent?: string }): Promise<LoginResult> {
    try {
      // 验证码开关：CAPTCHA_ENABLED=true 时强制校验（防爆破）
      if (this.configService.get<string>('CAPTCHA_ENABLED') === 'true') {
        await this.captchaService.assertValid(dto.captchaId, dto.captchaCode)
      }

      const user = await this.userService.findByUsername(dto.username)
      if (!user) {
        throw new BizException(BizErrorCode.LOGIN_FAILED, '用户名或密码错误', HttpStatus.UNAUTHORIZED)
      }

      // JeecgBoot 兼容校验：PBEWithMD5AndDES(username, password, salt) 与库中 16 位 hex 比对
      if (!this.userService.verifyPassword(user, dto.password)) {
        throw new BizException(BizErrorCode.LOGIN_FAILED, '用户名或密码错误', HttpStatus.UNAUTHORIZED)
      }

      if (user.status !== USER_STATUS.NORMAL) {
        throw new BizException(BizErrorCode.ACCOUNT_DISABLED, '账号已被冻结，请联系管理员', HttpStatus.FORBIDDEN)
      }

      await this.logService.recordLogin({ username: dto.username, status: 1, message: '登录成功', ip: meta.ip, userAgent: meta.userAgent })
      return this.issueTokens(user)
    } catch (error) {
      const message = error instanceof HttpException ? error.message : '登录异常'
      await this.logService.recordLogin({ username: dto.username, status: 0, message, ip: meta.ip, userAgent: meta.userAgent })
      throw error
    }
  }

  /** 用 refreshToken 换新令牌对 */
  async refreshToken(refreshToken: string): Promise<LoginResult> {
    let payload: JwtPayload
    try {
      payload = this.jwtService.verify<JwtPayload>(refreshToken, {
        secret: this.configService.get<string>('JWT_REFRESH_SECRET')
      })
    } catch {
      throw new BizException(BizErrorCode.UNAUTHORIZED, 'refreshToken 无效或已过期', HttpStatus.UNAUTHORIZED)
    }

    // 与登录返回保持同构：带角色与部门（depts / dept / deptId）
    const user = await this.userService.findOneDetail(String(payload.id))
    return this.issueTokens(user)
  }

  /** 签发 accessToken + refreshToken */
  private async issueTokens(user: import('../../user/entities/user.entity').UserEntity): Promise<LoginResult> {
    const payload: JwtPayload = { id: user.id, username: user.username }
    const secret = this.configService.get<string>('JWT_SECRET')
    const refreshSecret = this.configService.get<string>('JWT_REFRESH_SECRET')
    const expiresIn = (this.configService.get<string>('JWT_EXPIRES_IN') || '2h') as StringValue
    const refreshExpiresIn = (this.configService.get<string>('JWT_REFRESH_EXPIRES_IN') || '7d') as StringValue

    const [accessToken, refreshToken] = await Promise.all([
      this.jwtService.signAsync(payload, { secret, expiresIn }),
      this.jwtService.signAsync(payload, { secret: refreshSecret, expiresIn: refreshExpiresIn })
    ])

    return {
      accessToken,
      refreshToken,
      expiresIn: this.parseExpiresIn(expiresIn),
      userInfo: this.userService.stripPassword(user)
    }
  }

  /** 把 "2h"/"7d" 等过期时间换算成秒，方便前端定时刷新 */
  private parseExpiresIn(value: string): number {
    const match = /^(\d+)([smhd])$/.exec(value)
    if (!match) return 7200
    const multipliers: Record<string, number> = { s: 1, m: 60, h: 3600, d: 86400 }
    return Number(match[1]) * multipliers[match[2]]
  }
}

@Module({

        signOptions: { expiresIn: (configService.get<string>('JWT_EXPIRES_IN') || '2h') as StringValue }
      })
    })
  ],
  controllers: [AuthController, AuthV1Controller],
  providers: [AuthService, CaptchaService, JwtStrategy, JwtAuthGuard],
  exports: [AuthService, JwtAuthGuard]
})
export class AuthModule {}
```

## 业务码速查表（前端对照用）

| code | 含义 | HTTP 状态 | 触发场景 |
|---|---|---|---|
| 0 | 成功 | 200 | 所有正常返回 |
| 412 | 验证码错误 | 200 | 验证码内容对不上 |
| 413 | 验证码已过期 | 200 | 超过 5 分钟或 captchaId 从未生成 |
| 415 | 验证码已被使用 | 200 | 并发抢用同一验证码 |
| 417 | 验证码参数错误 | 200 | 没传 captchaId / captchaCode |
| 503 | 依赖服务不可用 | 200 | Redis 未启用，验证码不可用 |
| 1001 | 参数校验失败 | 400 | class-validator 校验不过（message 取第一条） |
| 1002 | 资源不存在 | 404 | 用户/角色/部门等查不到 |
| 1003 | 未授权 | 401 | 没带 token、token 过期、refreshToken 无效 |
| 1004 | 禁止访问 | 403 | 角色/按钮权限不足 |
| 1005 | 业务处理失败 | 200 | 部门不存在、保护内置账号等业务规则 |
| 1007 | 登录失败 | 401 | 用户名或密码错误 |
| 1008 | 账号停用 | 403 | 登录时账号非正常状态 |
| 1006 | 服务器内部错误 | 500 | 未捕获异常兜底 |

## 遇到的坑

1. **「错误验证码」实测先返回 413 而不是 412**：因为正确登录已把该验证码消费掉（一次性），同一个 captchaId 再查 Redis 值为空 → 走「已过期」分支。这不是 bug，是校验顺序「值存在性 → 是否被并发消费 → 内容比对」的必然结果；单独测 412 必须重新取一个验证码、故意填错。
2. **`assertValid` 里 GET+DEL 必须放 MULTI 事务**：先 GET 再 DEL 两条独立命令之间，并发请求可能同时读到值（都能通过比对），一次性就被打穿了；`exec()` 返回的第二个结果是 DEL 的受影响行数，为 0 即「刚被别人消费」，报 415。
3. **BizException 默认 HTTP 200 的取舍**：用户模板里默认是 `HttpStatus.BAD_REQUEST`，但本项目此前约定「业务失败走 HTTP 200 + code 区分」（前端 axios 不进 catch），所以默认值改成 `HttpStatus.OK`，登录失败等需要保留 401/403 语义的场景显式传第三个参数——**行为差异要写清楚，避免接前端时误判**。
4. **删除 `BusinessException` 前必须全库搜引用**：`grep -rn "BusinessException" src` 确认 user.service 的 2 处替换完再删文件，否则 build 报模块找不到。
5. **过滤器里英文默认提示要本地化**：Nest 的 `UnauthorizedException('...')` 不传参时 message 是 `Unauthorized`，前端直接弹英文不友好；`localize()` 按状态码映射中文，但**有自定义文案时必须原样保留**（判断条件是 `message === DEFAULT_EN_MESSAGES[status]` 而不是非空判断）。

## 怎么扩展 / 使用

- **新业务错误**：先在 `BizErrorCode` 枚举加值（注释写清楚），然后 `throw new BizException(BizErrorCode.XXX, '中文提示')`，过滤器自动组装响应，不用碰过滤器。
- **需要非 200 HTTP 语义**（如要触发 axios catch）：`throw new BizException(code, msg, HttpStatus.NOT_FOUND)`。
- **其他模块加 v1 路由**：照抄 `auth.v1.controller.ts` 的模式——新建 `xxx.v1.controller.ts`，`@Controller('v1/xxx')`，在对应 Module 的 `controllers` 数组注册，Service 复用不动。
- **前端切换**：baseURL 从 `/api` 改 `/api/v1` 即可用新路由；响应判断建议 `res.success` 或 `res.code === 0`，验证码错误特判 `code === 412`（弹提示 + 刷新验证码图）。
